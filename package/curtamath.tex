% Partial derivative
\ExplSyntaxOn

% Define local variables
\clist_new:N \l_pdv_vars_clist  % List of partial-derivative variables
\clist_new:N \l_pdv_exps_clist  % List of partial-derivative exponents
\int_new:N   \l_pdv_index_int   % Partial-derivative index
\tl_new:N    \l_pdv_exp_tl      % New tokenlist (string block) of current exponent

% Usage: \pdv{f}[order]{variables}[exponents]
% #1: Function (Mandatory)
% #2: Total order (Optional, default: empty)
% #3: Variable list (Mandatory)
% #4: Exponents list (Optional, default: empty)
\NewDocumentCommand{\pdv}{ m O{} m O{} }
{
	\frac
	{
		\partial \tl_if_blank:nF {#2} { ^{#2} } {#1}  % If token list of #2 is empty, prints nothing
	}
	{
		\clist_set:Nn \l_pdv_vars_clist {#3}  % Passes #3 to comma-separated list of variables
		\clist_set:Nn \l_pdv_exps_clist {#4}  % Passes #4 to comma-separated list of exponents
		\int_zero:N \l_pdv_index_int 		  % Reading index
		
		\clist_map_inline:Nn \l_pdv_vars_clist  % For each element in variables list
		{
			\int_incr:N \l_pdv_index_int  % Increments reading index
			
			\tl_set:Nx \l_pdv_exp_tl  % Sets the exponent as the index-element of exponent list
			{ \clist_item:Nn \l_pdv_exps_clist { \l_pdv_index_int } }
			
			\partial ##1
			\bool_if:nT  % If boolean is empty or equal to 1, skip
			{
				! \tl_if_blank_p:V \l_pdv_exp_tl
				&& ! \str_if_eq_p:Vn \l_pdv_exp_tl { 1 }
			}
			{ ^{\l_pdv_exp_tl} }
		}
	}
}

\ExplSyntaxOff
